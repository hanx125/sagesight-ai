def get_recommended_preset():
    """
    根据系统配置推荐最优硬件加速预设

    Returns:
        str: 推荐的预设名称
    """
    system = platform.system()

    if system == "Windows":
        # Windows 平台优先级：NVIDIA > Intel QSV > DXVA2
        try:
            import subprocess
            # 检查是否有 NVIDIA GPU
            result = subprocess.run(
                ['nvidia-smi'],
                capture_output=True,
                text=True,
                timeout=2
            )
            if result.returncode == 0:
                return "preset-nvidia", "NVIDIA CUDA（推荐，性能最优）"
        except:
            pass

        # 如果没有 NVIDIA GPU，使用 DXVA2
        return "preset-dxva2", "DXVA2（Windows 默认）"

    elif system == "Linux":
        # Linux 平台优先级：NVIDIA > AMD VAAPI > Intel QSV
        try:
            import subprocess
            # 检查是否有 NVIDIA GPU
            result = subprocess.run(
                ['nvidia-smi'],
                capture_output=True,
                text=True,
                timeout=2
            )
            if result.returncode == 0:
                return "preset-nvidia", "NVIDIA CUDA（推荐）"
        except:
            pass

        # 检查是否有 AMD GPU（VAAPI）
        if os.path.exists("/dev/dri"):
            return "preset-vaapi", "AMD VAAPI"

        # 默认使用 CPU 解码
        return "", "CPU 解码（无硬件加速）"

    else:
        return "", "CPU 解码（不支持硬件加速）"
